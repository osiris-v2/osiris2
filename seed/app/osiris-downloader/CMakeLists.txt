cmake_minimum_required(VERSION 3.10)
project(OsirisDownloader C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)          # GNU C: computed goto, cleanup
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET sdl2 SDL2_ttf)
find_package(Threads REQUIRED)
add_executable(osiris-downloader src/main.c src/core.c src/ui.c)
target_compile_options(osiris-downloader PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(osiris-downloader PRIVATE PkgConfig::DEPS Threads::Threads)
set_target_properties(osiris-downloader PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
